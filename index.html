// VIEW const viewButton=document.createElement("button"); viewButton.textContent="VIEW"; viewButton.onclick=()=>viewMaterial(item.file_path); entry.appendChild(viewButton); // DOWNLOAD FOR ADMIN AND OFFICES if(role!=="teacher"){ const downloadButton=document.createElement("button");
