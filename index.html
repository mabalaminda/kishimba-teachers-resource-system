import { createClient } from "https://esm.sh/@supabase/supabase-js@2"; const corsHeaders = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type", "Access-Control-Allow-Methods": "POST, OPTIONS", }; Deno.serve(async (req) => { if (req.method === "OPTIONS") { return new Response("ok", { headers: corsHeaders }); } const respond = (data: unknown, status = 200) => new Response(JSON.stringify(data), { status, headers: { ...corsHeaders, "Content-Type": "application/json", }, }); const supabaseUrl = Deno.env.get("SUPABASE_URL"); const serviceKey = Deno.env.get("SERVICE_ROLE_KEY"); if (!supabaseUrl || !serviceKey) { return respond({ error: "Supabase secrets hazijakamilika." }, 500); } const authHeader = req.headers.get("Authorization"); if (!authHeader?.startsWith("Bearer ")) { return respond({ error: "Huruhusiwi. Ingia kwanza." }, 401); } const supabase = createClient(supabaseUrl, serviceKey, { auth: { autoRefreshToken: false, persistSession: false, }, }); let teacherId: string | null = null; let teacherCreated = false; let roleCreated = false; let assignmentsCreated = false; let subjectsCreated = false; try { const token = authHeader.slice(7); const { data: authData, error: authError } = await supabase.auth.getUser(token); if (authError || !authData.user) { return respond({ error: "Taarifa za kuingia si sahihi." }, 401); } const { data: callerRole, error: roleError } = await supabase .from("user_roles") .select("role") .eq("user_id", authData.user.id) .maybeSingle(); if ( roleError || !callerRole || !["Administrator", "Headteacher", "Academic"].includes( String(callerRole.role) ) ) { return respond({ error: "Huna ruhusa ya kusajili walimu." }, 403); } const body = await req.json(); const full_name = String(body.full_name || "").trim(); const email = String(body.email || "").trim().toLowerCase(); const phone = String(body.phone || "").trim(); const academic_year = String( body.academic_year || new Date().getFullYear() ); const subjects = Array.isArray(body.subjects) ? body.subjects.map((s: unknown) => String(s).trim()).filter(Boolean) : String(body.subjects || "") .split(",") .map((s: string) => s.trim()) .filter(Boolean); const assignments = Array.isArray(body.assignments) ? body.assignments : []; if (!full_name || !email) { return respond({ error: "Jina kamili na email vinahitajika.", }, 400); } if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) { return respond({ error: "Email si sahihi." }, 400); } if (!subjects.length) { return respond({ error: "Weka angalau somo moja." }, 400); } if (!assignments.length) { return respond({ error: "Weka angalau darasa moja au combination moja.", }, 400); } const validForms = [ "Form One", "Form Two", "Form Three", "Form Four", "Form Five", "Form Six", ]; const validStreams = ["A", "B", "C", "D", "E", "F"]; for (const item of assignments) { if ( !item || !["O-Level", "A-Level"].includes(item.education_level) || !validForms.includes(item.form_level) ) { return respond({ error: "Taarifa za darasa si sahihi." }, 400); } if (item.education_level === "O-Level") { if ( !["Form One", "Form Two", "Form Three", "Form Four"].includes( item.form_level ) || !validStreams.includes(item.stream) || item.combination ) { return respond({ error: "O-Level inahitaji Form One hadi Four na stream A hadi F.", }, 400); } } if (item.education_level === "A-Level") { if ( !["Form Five", "Form Six"].includes(item.form_level) || !String(item.combination || "").trim() || item.stream ) { return respond({ error: "A-Level inahitaji Form Five au Six na combination.", }, 400); } } } const { data: existing, error: existingError } = await supabase .from("teachers") .select("id") .eq("email", email) .maybeSingle(); if (existingError) { return respond({ error: existingError.message }, 400); } if (existing) { return respond({ error: "Email hii tayari imesajiliwa." }, 409); } const redirectTo = "https://shimmering-gingersnap-5c2932.netlify.app/"; const { data: inviteData, error: inviteError } = await supabase.auth.admin.generateLink({ type: "invite", email, options: { redirectTo, data: { full_name }, }, }); if ( inviteError || !inviteData?.user || !inviteData?.properties?.action_link ) { return respond({ error: inviteError?.message || "Imeshindwa kutengeneza link.", }, 400); } teacherId = inviteData.user.id; const { error: teacherError } = await supabase .from("teachers") .insert({ id: teacherId, user_id: teacherId, full_name, email, phone: phone || null, subject: subjects.join(", "), role: "teacher", }); if (teacherError) { throw new Error("Teachers: " + teacherError.message); } teacherCreated = true; const { error: userRoleError } = await supabase .from("user_roles") .insert({ user_id: teacherId, role: "Teacher", }); if (userRoleError) { throw new Error("User roles: " + userRoleError.message); } roleCreated = true; const assignmentRows = assignments.map((item: any) => ({ teacher_id: teacherId, education_level: item.education_level, form_level: item.form_level, stream: item.education_level === "O-Level" ? item.stream : null, combination: item.education_level === "A-Level" ? String(item.combination).trim().toUpperCase() : null, })); const { error: assignmentError } = await supabase .from("teacher_assignments") .insert(assignmentRows); if (assignmentError) { throw new Error("Teacher assignments: " + assignmentError.message); } assignmentsCreated = true; const subjectRows = assignments.flatMap((item: any) => subjects.map((subject: string) => ({ teacher_id: teacherId, subject, form_level: item.form_level, education_level: item.education_level, combination: item.education_level === "A-Level" ? String(item.combination).trim().toUpperCase() : null, academic_year, is_active: true, })) ); const { error: subjectError } = await supabase .from("teacher_subject_assignments") .insert(subjectRows); if (subjectError) { throw new Error( "Teacher subject assignments: " + subjectError.message ); } subjectsCreated = true; return respond({ success: true, message: "Mwalimu amesajiliwa pamoja na madarasa na masomo yake.", teacher_id: teacherId, invite_link: inviteData.properties.action_link, }); } catch (error) { if (teacherId) { if (subjectsCreated) { await supabase .from("teacher_subject_assignments") .delete() .eq("teacher_id", teacherId); } if (assignmentsCreated) { await supabase .from("teacher_assignments") .delete() .eq("teacher_id", teacherId); } if (roleCreated) { await supabase .from("user_roles") .delete() .eq("user_id", teacherId); } if (teacherCreated) { await supabase .from("teachers") .delete() .eq("id", teacherId); } await supabase.auth.admin.deleteUser(teacherId); } return respond({ error: error instanceof Error ? error.message : "Hitilafu imetokea.", }, 500); } });
